# ============================================================
# Program 01 — Comparison Measure for the Parametric Framework
# ============================================================
# This program finds the optimal 3-interval arcsine measure ρ 
# of mass λ = 37/40 that satisfies the potential bound:
#     max_{t≥0} [2U^ρ(t) - V(t)] ≤ -6.0
#
# IMPORTANT: In the paper, we proved that the external field 
# V_s(t) has the coefficient ν=2 (independent of s). Therefore, 
# this single measure works for ALL odd 7 <= s <= 49.
#
# Outputs the optimal L_j and c_j to be used in Programs 02 and 03.
# Runtime: ~30-60 seconds.
# ============================================================

from sage.all import *
import numpy as np
from scipy.optimize import minimize

# ---------- Parameters ----------
alpha = 0.075
lam   = 0.925
nu    = 2     # Fixed pole constant, independent of s

# ---------- Grid for max evaluation ----------
t_grid = np.concatenate([
    np.linspace(1e-6, 0.5, 400),
    np.linspace(0.5, 3, 400),
    np.linspace(3, 15, 100),
])
N_t = len(t_grid)

# ---------- Mathematical building blocks ----------
def J_a_vec(a, t):
    """J_a(t) = a log(t + a^2) - 2a + 2√t arctan(a/√t)"""
    return a*np.log(t + a**2) - 2*a + 2*np.sqrt(t)*np.arctan(a/np.sqrt(t))

def U_arc_vec(L, t):
    """Potential of the arcsine measure on [0, L]."""
    L = max(L, 1e-9)
    result = np.where(t <= L, np.log(L/4), 0.0)
    mask = t > L
    if mask.any():
        tt = t[mask]
        result[mask] = np.log((tt - L/2 + np.sqrt(tt*(tt-L)))/2)
    return result

def V_t_full(t):
    """External field V(t) with ν = 2 (independent of s)."""
    return 2*np.pi*np.sqrt(t) + J_a_vec(1, t) - nu*J_a_vec(alpha, t)

V_t = V_t_full(t_grid)

def M_rho(Ls, cs):
    """Compute M_0 = max_t [2U^ρ(t) - V(t)]."""
    n = len(Ls)
    U = np.zeros(N_t)
    for j in range(n):
        U += cs[j] * U_arc_vec(Ls[j], t_grid)
    return np.max(2*U - V_t)

def I_rho(Ls, cs):
    """Compute the logarithmic energy I(ρ) of the arcsine measure."""
    n = len(Ls)
    I = 0.0
    for i in range(n):
        for j in range(n):
            I += cs[i]*cs[j]*np.log(max(Ls[i], Ls[j])/4)
    return I

def C_star():
    """The constant correction C_* from the asymptotic analysis."""
    return (-2*lam
            + 2*nu*alpha*lam*(1 - np.log(alpha))
            + 3*lam**2
            - 2*lam**2*np.log(2*lam))

# ---------- Objective with penalty for M_0 > -6.0 ----------
n_int = 3

def obj(x):
    Ls = np.exp(x[:n_int])
    raw = np.exp(x[n_int:])
    cs = raw / raw.sum() * lam
    M = M_rho(Ls, cs)
    # Penalize if M_0 > -6.0 to enforce the potential bound
    pen = 100.0 * max(0.0, M + 6.0)**2
    return lam*M - I_rho(Ls, cs) + C_star() + pen

# ---------- 20 restarts of Nelder-Mead ----------
np.random.seed(42)
best = None
n_restart = 20

print(f"Running {n_restart} restarts of Nelder-Mead...")
for trial in range(n_restart):
    # Random initial guess in log-space to ensure positivity
    x0 = np.concatenate([
        np.log(np.sort(np.random.rand(n_int)*1.5 + 0.1)),
        np.log(np.random.rand(n_int) + 0.1)
    ])
    res = minimize(obj, x0, method='Nelder-Mead',
                   options={'maxiter': 2000, 'xatol': 1e-7, 'fatol': 1e-7})
    if best is None or res.fun < best.fun:
        best = res
        
    Ls_try = np.exp(res.x[:n_int])
    raw_try = np.exp(res.x[n_int:])
    cs_try = raw_try / raw_try.sum() * lam
    M_try = M_rho(Ls_try, cs_try)
    marker = " *" if best is res else ""
    print(f"  restart {trial+1:2d}/{n_restart}: M_0 = {M_try:+.6f}{marker}")

# ---------- Extract optimal ----------
Ls = np.exp(best.x[:n_int])
raw = np.exp(best.x[n_int:])
cs = raw / raw.sum() * lam

# Sort by increasing L
idx = np.argsort(Ls)
Ls = Ls[idx]
cs = cs[idx]

# ---------- Report ----------
print()
print("=" * 62)
print("Program 01 — Comparison Measure for All s in [7, 49]")
print("=" * 62)
print(f"ν = {nu} (independent of s)   α = {alpha}   λ = {lam}")
print("-" * 62)
print("Optimal measure (copy these into Programs 02, 03, 04):")
for k in range(n_int):
    print(f"  L_{k+1} = {Ls[k]:.9f}   c_{k+1} = {cs[k]:.9f}")
print("-" * 62)
print(f"Mass ∑ c_j         = {sum(cs):.12f}   (target 0.925)")
print(f"M_0 = max(2U-V)    = {M_rho(Ls, cs):.12f}")
print(f"I(ρ)               = {I_rho(Ls, cs):.12f}")
print(f"C_*                = {C_star():.12f}")
print(f"U = λM_0 - I + C*  = {lam*M_rho(Ls, cs) - I_rho(Ls, cs) + C_star():.12f}")
print("-" * 62)

M0_final = M_rho(Ls, cs)
I_final = I_rho(Ls, cs)
C_final = C_star()
U_final = lam*M0_final - I_final + C_final

# ---------- Sanity checks ----------
checks = {
    "Mass = λ":   abs(sum(cs) - lam) < 1e-10,
    "M_0 ≤ -6.0": M0_final <= -6.0,
    "U < 0":      U_final < 0,
}
for name, ok in checks.items():
    print(f"  {name:<15}: {ok}")

ok_all = all(checks.values())
print(f"==> {'PASS' if ok_all else 'FAIL'} (Program 01)")
print("=" * 62)
